					\item $^{60}\mathrm{Co}$放射源：主峰2峰位：771，对应能量：1.173MeV

	\subsubsection{实验二 \quad 计算$^{137}\mathrm{Cs}$的康普顿散射峰和背散射峰能量与测量值的相对误差}

		\begin{enumerate}
			\item 理论计算得到：
				\begin{itemize}
					\item 康普顿散射峰能量：$E_{max} = 0.47765 \ \mathrm{(MeV)}$
					\item 康普顿背散射峰能量：$E_b = E_{\gamma} - E_{max} = 0.18345 \ \mathrm{(MeV)}$
				\end{itemize}

			\item 实验测量得到：
				\begin{enumerate}
					\item 康普顿散射峰峰位：289 $\Rightarrow E'_{max} = 0.0015 \times 289 - 0.0005 \ \mathrm{(MeV)} = 0.433 \ \mathrm{(MeV)}$
					\item 康普顿背散射峰峰位：141 $\Rightarrow E'_b = 0.0015 \times 141 - 0.0005 \ \mathrm{(MeV)} = 0.211 \ \mathrm{(MeV)}$
				\end{enumerate}

			\item 实验值与理论值的相对误差为：
				\begin{enumerate}
					\item $\eta_{E_{max}} = \frac{0.433 - 0.47765}{0.47765} \times 100 \% =  -9.35 \% $
